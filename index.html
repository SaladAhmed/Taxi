function parseBankPDFText(fullText) { const user = auth.currentUser; if (!user) return; let lines = fullText.split('\n'); let aggregatedEntries = {}; // Shaygan wuxuu isku geynayaa (total) transaksjonada isku midka ah // RegEx-ka DNB ee raadinaya Taariikhda (DD.MM.YY) iyo qoraalka la socda const dnbRegex = /^(\d{2})\.(\d{2})\.(\d{2})\s*,\s*([\s\S]*?)([\d\s]+,\d{2})/; const dnbRegexPage2 = /^(\d{2})\.(\d{2})\.(\d{2})\s*([\s\S]*?)([\d\s]+,\d{2})/; lines.forEach(line => { let cleanLine = line.trim(); let date = "", text = "", rawAmount = ""; let match1 = cleanLine.match(dnbRegex); if (match1) { date = match1[1] + "." + match1[2] + "." + match1[3]; text = match1[4].replace(/\s+/g, ' ').trim(); rawAmount = match1[5].trim(); } else { let match2 = cleanLine.match(dnbRegexPage2); if (match2) { date = match2[1] + "." + match2[2] + "." + match2[3]; text = match2[4].replace(/\s+/g, ' ').trim(); rawAmount = match2[5].trim(); } } // Haddii foomku sax yahay, bilow kala saarista qaybaha (Category Matching) if (date && text && rawAmount && rawAmount !== "0,00") { let lineLower = text.toLowerCase(); let targetCat = null; if (lineLower.includes("agder taxi")) { targetCat = "3030"; } else if (lineLower.includes("tesla") || lineLower.includes("uno-x") || lineLower.includes("eviny")) { targetCat = "7000"; // Drivstoff / Lading } else if (lineLower.includes("nextgentel") || lineLower.includes("lyse tele")) { targetCat = "6907"; } else if (lineLower.includes("fortum") || lineLower.includes("vibb")) { targetCat = "6340"; } else if (lineLower.includes("forsikring") || lineLower.includes("dnb livsforsikring")) { targetCat = "7500"; } else if (lineLower.includes("omkostninger")) { targetCat = "7770"; } else if (lineLower.includes("santander")) { targetCat = "5000"; } else if (lineLower.includes("skatteetaten") || lineLower.includes("skatt/arbeidsavgift")) { targetCat = "5400"; } // Kaliya haddii la helay Account u dhigma (Target Category) if (targetCat) { // Nadiifinta lacagta nidaamka Norway (Tusaale: 107 831,84 -> 107831.84) let cleanAmt = rawAmount.replace(/\s/g, '').replace(/\./g, '').replace(',', '.'); let parsedAmt = parseFloat(cleanAmt); if (!isNaN(parsedAmt) && parsedAmt > 0) { // Habaynta taariikhda (DD.MM.YY -> YYYY-MM-DD) let dateParts = date.split('.'); let formattedDate = `20${dateParts[2]}-${dateParts[1]}-${dateParts[0]}`; // Furaha isku geynta: Safka wuxuu noqonayaa hal saf haddii Taariikhda iyo Account-ku isku mid yihiin let groupKey = `${formattedDate}_${targetCat}`; if (!aggregatedEntries[groupKey]) { // Haddii uusan furehan hore u jirin, abuuro let defaultDesc = text; if (targetCat === "7000") defaultDesc = "Drivstoff / Lading (Tesla / Uno-X)"; aggregatedEntries[groupKey] = { desc: defaultDesc, amt: 0, // Waxaa lagu dari doonaa hoos subCat: targetCat, mainCat: metaAccounts[targetCat].type, // 'income' ama 'expense' -> Tani waxay xisaabisaa gaduud/cagaar date: formattedDate }; } // Halkan ayay lacagtu isku dhex girmaysaa (Totaling) aggregatedEntries[groupKey].amt += parsedAmt; } } } }); // U beddel shaygii (Object) guud u ahaa Array si loogu diro Firebase let finalEntries = Object.values(aggregatedEntries); if (finalEntries.length === 0) { showStatus("Wax xog ah oo la aqoonsaday lagama helin PDF-ka rasmiga ah!", "error"); return; } // Shubista Firebase (Tirtir xogtii hore ee maalintaas, kuna beddel tan cusub oo la isku geeyay) const userRef = db.ref('users/' + user.uid + '/entries'); userRef.remove().then(() => { let promises = finalEntries.map(e => userRef.push(e)); Promise.all(promises).then(() => { showStatus(`Guul! ${finalEntries.length} Xogood ayaa si sax ah loo habeeyay! ✅`, "success"); alert(`Xogtaadii waa la hagaajiyay! Waxaa la helay ${finalEntries.length} transaksjoner oo isku geysan.`); showPg('dash'); if (typeof loadData === "function") loadData(); }); }); } // ---- SHAQADA CSV-GA OO CUSUB (Si aad u baddesho badhanka in-active-ka ahaa) ---- function exportTableToCSV(filename = 'transport_pro_transactions.csv') { let csv = []; let rows = document.querySelectorAll("table tr"); // Wuxuu dhex raadinayaa shaxda dashboard-ka kor ku xusan for (let i = 0; i < rows.length; i++) { let row = [], cols = rows[i].querySelectorAll("td, th"); for (let j = 0; j < cols.length; j++) { // Ka saar wixii xariiq dhexda uga jira ama calaamado khalkhal gelinaya CSV let data = cols[j].innerText.replace(/(\r\n|\n|\r)/gm, "").replace(/;/g, ","); row.push(`"${data}"`); } csv.push(row.join(";")); // Semicolon waa nidaamka Excel ee Norway u doorbido } // Soo bixi faylka CSV-ga ah iyadoo la siinayo habka UTF-8 Bom si loogu akhriyo xarfaha ÆØÅ let csvFile = new Blob([new Uint8Array([0xEF, 0xBB, 0xBF]), csv.join("\n")], {type: "text/csv;charset=utf-8;"}); let downloadLink = document.createElement("a"); downloadLink.download = filename; downloadLink.href = window.URL.createObjectURL(csvFile); downloadLink.style.display = "none"; document.body.appendChild(downloadLink); downloadLink.click(); }
