function parseBankPDFText(fullText) { const user = auth.currentUser; if (!user) return; let lines = fullText.split('\n'); let finalEntries = []; // RegEx aad u saxan oo u goonni ah qaabka PDF-ka DNB // Wuxuu raadinayaa Taariikh (DD.MM.YY) ku xigto Qoraal, ka dibna Lacag dhibic/kooma leh const dnbRegex = /^"(\d{2}\.\d{2}\.\d{2})"\s*,\s*"(\d{2}\.\d{2}\.\d{2})"\s*,\s*"([^"]+)"\s*,\s*"([^"]*)"\s*,\s*"([^"]*)"/; const dnbRegexPage2 = /^"(\d{2}\.\d{2}\.\d{2})"\s*,\s*"([^"]+)"\s*,\s*"([^"]*)"\s*,\s*"([^"]*)"/; lines.forEach(line => { let cleanLine = line.trim(); let date = "", text = "", utAvKonto = "", innPaKonto = ""; // Isku day qaabka bogga 1-aad (oo leh labo taariikhrood oo isku xiga) let match1 = cleanLine.match(dnbRegex); if (match1) { date = match1[2]; // Isticmaal taariikhda bokføring-ka (DD.MM.YY) text = match1[3].replace(/\s+/g, ' ').trim(); utAvKonto = match1[4].trim(); innPaKonto = match1[5].trim(); } else { // Isku day qaabka bogga 2-aad/3-aad (oo mararka qaar hal taariikh ka muuqato bilowga) let match2 = cleanLine.match(dnbRegexPage2); if (match2) { date = match2[1]; text = match2[2].replace(/\s+/g, ' ').trim(); utAvKonto = match2[3].trim(); innPaKonto = match2[4].trim(); } } // Haddii aan helay taariikh iyo qoraal sax ah, aynu guda galno sifaynta qaybaha if (date && text) { let lineLower = text.toLowerCase(); let targetCat = null; // Kala saarista qaybaha (Category Matching) if (lineLower.includes("agder taxi")) targetCat = "3030"; else if (lineLower.includes("tesla") || lineLower.includes("uno-x") || lineLower.includes("eviny")) targetCat = "7000"; else if (lineLower.includes("nextgentel") || lineLower.includes("lyse tele")) targetCat = "6907"; else if (lineLower.includes("fortum") || lineLower.includes("vibb")) targetCat = "6340"; else if (lineLower.includes("forsikring") || lineLower.includes("dnb livsforsikring")) targetCat = "7500"; else if (lineLower.includes("omkostninger")) targetCat = "7770"; else if (lineLower.includes("santander")) targetCat = "5000"; else if (lineLower.includes("skatteetaten") || lineLower.includes("skatt/arbeidsavgift")) targetCat = "5400"; if (targetCat) { // Dooro lacagta saxda ah iyada oo la raacayo nidaamka xisaabaadka (Income vs Expense) let rawAmount = (metaAccounts[targetCat].type === "income") ? innPaKonto : utAvKonto; if (rawAmount && rawAmount !== "0,00") { // Nadiifinta lacagta Noorwiijiga (Tusaale: 34.663,00 -> 34663.00) let cleanAmt = rawAmount.replace(/\./g, '').replace(',', '.'); let parsedAmt = parseFloat(cleanAmt); if (!isNaN(parsedAmt) && parsedAmt > 0) { // Habaynta taariikhda saxda ah (DD.MM.YY -> YYYY-MM-DD) let dateParts = date.split('.'); let formattedDate = `20${dateParts[2]}-${dateParts[1]}-${dateParts[0]}`; finalEntries.push({ desc: text, amt: parsedAmt, subCat: targetCat, mainCat: metaAccounts[targetCat].type, date: formattedDate }); } } } } }); // Shubista Firebase (Tirtir xogtii hore ee khaldanayd, kuna beddel tan cusub) if (finalEntries.length === 0) { showStatus("Wax xog ah oo la aqoonsaday lagama helin PDF-ka rasmiga ah!", "error"); return; } const userRef = db.ref('users/' + user.uid + '/entries'); userRef.remove().then(() => { let promises = finalEntries.map(e => userRef.push(e)); Promise.all(promises).then(() => { showStatus(`Guul! ${finalEntries.length} Xogood ayaa si sax ah loo habeeyay! ✅`, "success"); alert(`Xogtaadii waa la hagaajiyay! Waxaa la helay ${finalEntries.length} transaksjoner.`); showPg('dash'); if (typeof loadData === "function") loadData(); }); }); }
